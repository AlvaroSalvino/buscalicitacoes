<VirtualHost *:443>
    ServerName licitacoes.meuapp.dev.br
    ServerAlias www.licitacoes.meuapp.dev.br
    ServerAdmin ti@licitacoes.meuapp.dev.br
    DocumentRoot /home/meu_app/public_html

    # Estáticos (collectstatic -> STATIC_ROOT) e mídia: antes do WSGIScriptAlias
    Alias /static /home/meu_app/public_html/buscalicitacoes/staticfiles
    <Directory /home/meu_app/public_html/buscalicitacoes/staticfiles>
        Require all granted
    </Directory>

    Alias /media /home/meu_app/public_html/buscalicitacoes/media
    <Directory /home/meu_app/public_html/buscalicitacoes/media>
        Options FollowSymLinks
        AllowOverride None
        Require all granted
    </Directory>

    <Directory /home/meu_app/public_html/buscalicitacoes/buscalicitacoes>
        <Files wsgi.py>
            Require all granted
        </Files>
    </Directory>

    WSGIDaemonProcess buscalicitacoes \
        python-home=/home/meu_app/public_html/buscalicitacoes/venv \
        python-path=/home/meu_app/public_html/buscalicitacoes \
        processes=2 threads=15 \
        header-buffer-size=131072 response-buffer-size=131072
    WSGIProcessGroup buscalicitacoes
    WSGIApplicationGroup %{GLOBAL}
    WSGIPassAuthorization On
    WSGIScriptAlias / /home/meu_app/public_html/buscalicitacoes/buscalicitacoes/wsgi.py

    SSLEngine on
    SSLCertificateFile    /etc/letsencrypt/live/licitacoes.meuapp.dev.br/fullchain.pem
    SSLCertificateKeyFile /etc/letsencrypt/live/licitacoes.meuapp.dev.br/privkey.pem

    ErrorLog ${APACHE_LOG_DIR}/buscalicitacoes_error.log
    CustomLog ${APACHE_LOG_DIR}/buscalicitacoes_access.log combined
